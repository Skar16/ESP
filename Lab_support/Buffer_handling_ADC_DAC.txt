#define LENGTH_DAC_BUF 1000

int16_t g_i16InBuf_R[LENGTH_ADC_BUF];
int16_t g_Count_DAC = 0;
int16_t g_i16OutBuf_L[LENGTH_DAC_BUF];
int16_t g_i16OutBuf_R[LENGTH_DAC_BUF];
